}

\newcommand{\FieldRow}[2]{%
  \vspace{6bp}%

    @{}>{\raggedright\arraybackslash\Aptos\fontsize{12bp}{18bp}\selectfont}X%
    >{\raggedleft\arraybackslash\Aptos\fontsize{12bp}{18bp}\selectfont}p{0.37\linewidth}@{}%
  }
  #1 & #2
  \end{tabularx}\par
}
