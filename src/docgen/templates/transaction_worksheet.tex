\newcommand{\SectionHeading}[1]{%
  \par\addvspace{0.95\baselineskip}%
  {\Aptos\bfseries\fontsize{12.5bp}{14.5bp}\selectfont\MakeUppercase{#1}\par}%

  {\color{ParamountBlue}\rule{\linewidth}{1.8bp}\par}%
  \vspace{3.2bp}%
}

\newcommand{\FieldRow}[2]{%
  \vspace{6bp}%
  \noindent
  \begin{tabularx}{\linewidth}{%
    @{}>{\raggedright\arraybackslash\Aptos\fontsize{12bp}{18bp}\selectfont}X%
    >{\raggedleft\arraybackslash\Aptos\fontsize{12bp}{18bp}\selectfont}p{0.37\linewidth}@{}%
  }
  \mbox{\MakeUppercase{#1}} & \mbox{#2}
  \end{tabularx}\par
}
