% Paragraphs and field rows share the space available to their column; there is
% no arbitrary per-field height limit. Headings retain their normal size.
\NewDocumentCommand{\FieldFont}{}
{
  \DocFont
  \fontsize
    { \fp_to_decimal:N \l__docgen_body_size_fp bp }
    { \fp_eval:n { 14 / 12 * \l__docgen_body_size_fp } bp }
  \selectfont
}

\NewDocumentCommand{\BodyLine}{+m}
{
  \group_begin:
  \DocFont
  \fontsize
    { \fp_to_decimal:N \l__docgen_body_size_fp bp }
    { \fp_eval:n { 1.5 * \l__docgen_body_size_fp } bp }
  \selectfont
  #1\par
  \group_end:
}

\NewDocumentCommand{\TermLine}{+m}
{
  \group_begin:
  \DocFont
  \fontsize
    { \fp_to_decimal:N \l__docgen_terms_size_fp bp }
    { \fp_eval:n { 1.5 * \l__docgen_terms_size_fp } bp }
  \selectfont
  #1\par
  \group_end:
}

\cs_new_protected:Npn \__docgen_measure_column:n #1
{
  \hbox_set:Nn \l__docgen_column_box
  {
    % Trial sizes must not leave false overflow warnings in the compiler log.
    % The final pass below restores normal diagnostics before typesetting.
    \hfuzz = \maxdimen
    \hbadness = 10000
    \begin{minipage}[t]{0.45\textwidth}
      \vspace{0pt}\raggedright
      #1
    \end{minipage}
  }
}

\NewDocumentCommand{\FittedColumn}{s m +m}
{
  \group_begin:
  \fp_set:Nn \l__docgen_body_size_fp { 12 }
  \fp_set:Nn \l__docgen_terms_size_fp { min(12, @@terms_font_ceiling@@) }
  \__docgen_measure_column:n { #3 }
  \bool_while_do:nn
    {
      \dim_compare_p:n
        { \box_ht:N \l__docgen_column_box + \box_dp:N \l__docgen_column_box > \WorksheetColumnHeight }
      && \fp_compare_p:n { \l__docgen_body_size_fp > 9 }
    }
  {
    \fp_sub:Nn \l__docgen_body_size_fp { 0.1 }
    \fp_set:Nn \l__docgen_terms_size_fp
      { min(\l__docgen_body_size_fp, @@terms_font_ceiling@@) }
    \__docgen_measure_column:n { #3 }
  }

  % Only the starred (terms) column may shrink paragraphs below 9 points.
  % Halving finds a fitting size quickly; bisection recovers the largest fit.
  \IfBooleanT{#1}
  {
    \dim_compare:nNnT
      { \box_ht:N \l__docgen_column_box + \box_dp:N \l__docgen_column_box }
      > { \WorksheetColumnHeight }
    {
      \fp_set_eq:NN \l__docgen_terms_high_fp \l__docgen_terms_size_fp
      \bool_while_do:nn
      {
        \dim_compare_p:n
          { \box_ht:N \l__docgen_column_box + \box_dp:N \l__docgen_column_box > \WorksheetColumnHeight }
        && \dim_compare_p:n
          { \fp_to_decimal:N \l__docgen_terms_size_fp bp > 1sp }
      }
      {
        % One scaled point is TeX's numeric resolution, not a readability floor.
        \fp_set:Nn \l__docgen_terms_size_fp { \l__docgen_terms_size_fp / 2 }
        \__docgen_measure_column:n { #3 }
      }
      \fp_set_eq:NN \l__docgen_terms_low_fp \l__docgen_terms_size_fp
      \prg_replicate:nn { 12 }
      {
        \fp_set:Nn \l__docgen_terms_size_fp
          { (\l__docgen_terms_low_fp + \l__docgen_terms_high_fp) / 2 }
        \__docgen_measure_column:n { #3 }
        \dim_compare:nNnTF
          { \box_ht:N \l__docgen_column_box + \box_dp:N \l__docgen_column_box }
          > { \WorksheetColumnHeight }
          { \fp_set_eq:NN \l__docgen_terms_high_fp \l__docgen_terms_size_fp }
          { \fp_set_eq:NN \l__docgen_terms_low_fp \l__docgen_terms_size_fp }
      }
      \fp_set_eq:NN \l__docgen_terms_size_fp \l__docgen_terms_low_fp
      \__docgen_measure_column:n { #3 }
    }
  }
  \dim_compare:nNnT
    { \box_ht:N \l__docgen_column_box + \box_dp:N \l__docgen_column_box }
    > { \WorksheetColumnHeight }
    {
      \typeout{DOCGEN_LAYOUT_ERROR:~#2}
      \PackageError{docgen}{Column~exceeds~space~at~minimum~font~size}
        {Shorten~long~fields~in~the~#2~column~and~generate~again.}
    }
  \begin{minipage}[t]{0.45\textwidth}
    \vspace{0pt}\raggedright
    #3
  \end{minipage}
  \group_end:
}
